cmake_minimum_required(VERSION 3.22)
project(android_dococr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ANDROID_OCR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
set(DOCOCR_SOURCE_ROOT "${ANDROID_OCR_ROOT}/native/docprase" CACHE PATH "docprase 源码快照")
set(MNN_SOURCE_ROOT "${ANDROID_OCR_ROOT}/native/MNN" CACHE PATH "固定版本 MNN 源码")
if(NOT EXISTS "${DOCOCR_SOURCE_ROOT}/CMakeLists.txt" OR NOT EXISTS "${MNN_SOURCE_ROOT}/CMakeLists.txt")
  message(FATAL_ERROR "缺少原生依赖，请在项目根目录运行 python tools/setup_native_deps.py")
endif()
foreach(flag MNN_BUILD_TOOLS MNN_BUILD_CONVERTER MNN_BUILD_DEMO MNN_LLM_BUILD_DEMO MNN_BUILD_TEST MNN_OPENCL MNN_VULKAN MNN_OPENGL MNN_OPENMP LLM_SUPPORT_HTTP_RESOURCE DOCOCR_BUILD_TESTS)
  set(${flag} OFF CACHE BOOL "Android 单图离线构建" FORCE)
endforeach()
option(MNN_SEP_BUILD "MNN 分库；关闭时将视觉和 LLM 合并到 MNN" ON)
set(MNN_BUILD_FOR_ANDROID_COMMAND ON CACHE BOOL "不执行 MNN Gradle 专用头文件复制" FORCE)
foreach(flag MNN_BUILD_SHARED_LIBS MNN_BUILD_LLM MNN_BUILD_OPENCV MNN_IMGCODECS MNN_LOW_MEMORY MNN_SUPPORT_TRANSFORMER_FUSE DOCOCR_REQUIRE_MNN DOCOCR_REQUIRE_LLM)
  set(${flag} ON CACHE BOOL "启用生产视觉/LLM 后端" FORCE)
endforeach()
add_subdirectory("${MNN_SOURCE_ROOT}" mnn EXCLUDE_FROM_ALL)
if(MNN_SEP_BUILD)
  # MNN 的 Android command 选项会把 Express 输出到 ../；恢复 Gradle 收集目录。
  set_target_properties(MNN_Express PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_LIBRARY_OUTPUT_DIRECTORY}")
endif()
# MNN_SEP_BUILD=OFF 时 llm 为 OBJECT target，其实现已合并入 MNN。
# 向 docprase 的现有查找接口传入实际 target，避免误连宿主 Linux .so。
set(DOCOCR_MNN_ROOT "${MNN_SOURCE_ROOT}" CACHE PATH "" FORCE)
set(DOCOCR_MNN_INCLUDE "${MNN_SOURCE_ROOT}/include" CACHE PATH "" FORCE)
set(DOCOCR_LLM_INCLUDE "${MNN_SOURCE_ROOT}/transformers/llm/engine/include" CACHE PATH "" FORCE)
set(DOCOCR_MNN_LIBRARY MNN CACHE STRING "" FORCE)
if(MNN_SEP_BUILD)
  set(DOCOCR_LLM_LIBRARY llm CACHE STRING "" FORCE)
else()
  set(DOCOCR_LLM_LIBRARY MNN CACHE STRING "" FORCE)
endif()
add_subdirectory("${DOCOCR_SOURCE_ROOT}" docprase EXCLUDE_FROM_ALL)
include("${CMAKE_CURRENT_LIST_DIR}/android_engine_sources.cmake")
# 桌面 PDF 渲染器依赖外部 Poppler/posix_spawn；保留 Android 原有 Java PDF 路径。
get_target_property(core_sources dococr_core SOURCES)
list(FILTER core_sources EXCLUDE REGEX "(^|/)pdf_renderer\\.cpp$")
set_property(TARGET dococr_core PROPERTY SOURCES "${core_sources}")
target_sources(dococr_core PRIVATE "${CMAKE_CURRENT_LIST_DIR}/pdf_renderer_android.cpp")
target_include_directories(dococr_core PRIVATE "${DOCOCR_SOURCE_ROOT}/src")
add_library(dococr_jni SHARED dococr_jni.cpp)
target_include_directories(dococr_jni PRIVATE "${DOCOCR_SOURCE_ROOT}/third_party/stb")
target_link_libraries(dococr_jni PRIVATE dococr_c)
target_compile_features(dococr_jni PRIVATE cxx_std_17)
