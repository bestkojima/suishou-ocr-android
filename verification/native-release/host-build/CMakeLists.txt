cmake_minimum_required(VERSION 3.22)
project(ocr_native_release_comparison LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
get_filename_component(OCR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(OCR_ADAPTER_ROOT "${OCR_ROOT}/app/src/main/cpp" CACHE PATH "本次或历史 Android 适配源码")
set(OCR_ENGINE_ROOT "${OCR_ROOT}/native/docprase" CACHE PATH "同一 docprase 源码快照")
set(MNN_SOURCE_ROOT "${OCR_ROOT}/native/MNN" CACHE PATH "固定版本 MNN 源码")
set(OCR_MNN_BUILD_ROOT "" CACHE PATH "可复用相同配置的 MNN 构建目录；不复用不同优化类型")
# APK 会 strip 符号；Debug 对照保留 O0/DEBUG/MNN_DEBUG，省略宿主 DWARF 以控制磁盘占用。
set(CMAKE_C_FLAGS_DEBUG "-g0" CACHE STRING "Debug 对照不保留 DWARF" FORCE)
set(CMAKE_CXX_FLAGS_DEBUG "-g0" CACHE STRING "Debug 对照不保留 DWARF" FORCE)
if(NOT OCR_MNN_BUILD_ROOT)
  foreach(flag MNN_BUILD_TOOLS MNN_BUILD_CONVERTER MNN_BUILD_DEMO MNN_LLM_BUILD_DEMO MNN_BUILD_TEST MNN_OPENCL MNN_VULKAN MNN_OPENGL MNN_OPENMP LLM_SUPPORT_HTTP_RESOURCE DOCOCR_BUILD_TESTS)
    set(${flag} OFF CACHE BOOL "同源 CPU 比较" FORCE)
  endforeach()
  foreach(flag MNN_BUILD_SHARED_LIBS MNN_SEP_BUILD MNN_BUILD_LLM MNN_BUILD_OPENCV MNN_IMGCODECS MNN_LOW_MEMORY MNN_SUPPORT_TRANSFORMER_FUSE LLM_SUPPORT_VISION)
    set(${flag} ON CACHE BOOL "与 Android 相同的模型后端" FORCE)
  endforeach()
  add_subdirectory("${MNN_SOURCE_ROOT}" mnn EXCLUDE_FROM_ALL)
  set(DOCOCR_MNN_LIBRARY MNN CACHE STRING "" FORCE)
  set(DOCOCR_LLM_LIBRARY llm CACHE STRING "" FORCE)
else()
  if(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "只允许 Release 对照复用 Release MNN")
  endif()
  file(STRINGS "${OCR_MNN_BUILD_ROOT}/CMakeCache.txt" mnn_build_type REGEX "^CMAKE_BUILD_TYPE:STRING=")
  if(NOT mnn_build_type STREQUAL "CMAKE_BUILD_TYPE:STRING=Release")
    message(FATAL_ERROR "复用的 MNN 必须确实为 Release")
  endif()
  set(DOCOCR_MNN_LIBRARY "${OCR_MNN_BUILD_ROOT}/mnn/libMNN.so" CACHE FILEPATH "" FORCE)
  set(DOCOCR_LLM_LIBRARY "${OCR_MNN_BUILD_ROOT}/mnn/libllm.so" CACHE FILEPATH "" FORCE)
endif()
set(DOCOCR_SOURCE_ROOT "${OCR_ENGINE_ROOT}")
set(DOCOCR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(DOCOCR_REQUIRE_MNN ON CACHE BOOL "" FORCE)
set(DOCOCR_REQUIRE_LLM ON CACHE BOOL "" FORCE)
set(DOCOCR_MNN_ROOT "${MNN_SOURCE_ROOT}" CACHE PATH "" FORCE)
set(DOCOCR_MNN_INCLUDE "${MNN_SOURCE_ROOT}/include" CACHE PATH "" FORCE)
set(DOCOCR_LLM_INCLUDE "${MNN_SOURCE_ROOT}/transformers/llm/engine/include" CACHE PATH "" FORCE)
add_subdirectory("${DOCOCR_SOURCE_ROOT}" docprase EXCLUDE_FROM_ALL)
include("${OCR_ADAPTER_ROOT}/android_engine_sources.cmake")
